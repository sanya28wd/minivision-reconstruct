{#- Shared template macros. Every value is escaped by the environment's autoescape. -#} {#- bilingual(key): an English + Arabic pair from the label catalogue (web/labels.json). The Arabic half is omitted, never invented, when the catalogue has no Arabic for the term. -#} {% macro bilingual(key) -%} {%- set item = label(key) -%} {{ item.en }} {%- if item.ar %} {{ item.ar }}{% endif -%} {%- endmacro %}
